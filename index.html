R Script R 语言全栈生信核心包一键加速安装 (清华/中科大源 + Bioconductor) 
# ==============================================================================
# Li Zhiyang (eto-1024) - R 生信核心包一键加速安装脚本
# ==============================================================================
options("repos" = c(CRAN = "https://mirrors.tuna.tsinghua.edu.cn/CRAN/"))
options(BioC_mirror = "https://mirrors.tuna.tsinghua.edu.cn/bioconductor")

if (!requireNamespace("BiocManager", quietly = TRUE)) {
    install.packages("BiocManager")
}

# 批量安装 CRAN + Bioconductor 核心生信与可视化套件
bio_packages <- c(
    "DESeq2", "edgeR", "limma", "Seurat", "SingleCellExperiment",
    "ComplexHeatmap", "clusterProfiler", "org.Hs.eg.db", "org.Mm.eg.db",
    "tidyverse", "ggplot2", "patchwork", "pheatmap", "survival", "survminer"
)

BiocManager::install(bio_packages, update = FALSE, ask = FALSE)
message("🎉 所有 R 语言核心生信包安装配置完成！")
